cmake_minimum_required(VERSION 3.16)
project(linkerbot-hand-tools VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()
include(CTest)
option(HAND_TOOLS_MODERN "Build the modern SDK adapter" OFF)
option(HAND_TOOLS_LEGACY "Build the legacy SDK adapter" OFF)
option(HAND_TOOLS_HISTORICAL_EXAMPLES "Build v2.2.1 hardware examples" OFF)
set(HAND_TOOLS_MODERN_SOURCE_DIR "" CACHE PATH "Modern SDK source root; empty uses an installed package")
set(HAND_TOOLS_LEGACY_ROOT "" CACHE PATH "Legacy SDK package root")
set(HAND_TOOLS_TRANSPORT_SOURCE_DIR "" CACHE PATH "Transport source root")

# The simulator/catalogue executable never links either SDK or a vendor runtime.
add_executable(hand_simulator src/bridge.cpp)
target_include_directories(hand_simulator PRIVATE include)
set_target_properties(hand_simulator PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(_config IN LISTS CMAKE_CONFIGURATION_TYPES)
    string(TOUPPER "${_config}" _upper)
    set_target_properties(hand_simulator PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${_upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_executable(hand_bridge src/bridge.cpp)
target_include_directories(hand_bridge PRIVATE include)
set_target_properties(hand_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(_config IN LISTS CMAKE_CONFIGURATION_TYPES)
    string(TOUPPER "${_config}" _upper)
    set_target_properties(hand_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${_upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(HAND_TOOLS_MODERN OR HAND_TOOLS_LEGACY)
    if(HAND_TOOLS_TRANSPORT_SOURCE_DIR)
        set(LINKERBOT_TRANSPORT_BUILD_TESTS OFF CACHE BOOL "")
        set(LINKERBOT_TRANSPORT_BUILD_EXAMPLES OFF CACHE BOOL "")
        set(LINKERBOT_TRANSPORT_ENABLE_SERIAL OFF CACHE BOOL "")
        set(LINKERBOT_TRANSPORT_ENABLE_MODBUS OFF CACHE BOOL "")
        add_subdirectory("${HAND_TOOLS_TRANSPORT_SOURCE_DIR}" _deps/transport)
    else()
        find_package(linkerbot-transport 0.3 CONFIG REQUIRED)
    endif()
    target_sources(hand_bridge PRIVATE src/transport.cpp)
    target_link_libraries(hand_bridge PRIVATE linkerbot::can)
endif()
if(HAND_TOOLS_MODERN)
    if(HAND_TOOLS_MODERN_SOURCE_DIR)
        set(LINKERBOT_BUILD_TESTS OFF CACHE BOOL "")
        set(LINKERBOT_BUILD_EXAMPLES OFF CACHE BOOL "")
        set(LINKERBOT_BUILD_TOOLS OFF CACHE BOOL "")
        add_subdirectory("${HAND_TOOLS_MODERN_SOURCE_DIR}" _deps/modern)
    else()
        find_package(linkerbot-hand-cpp 0.3 CONFIG REQUIRED)
    endif()
    target_sources(hand_bridge PRIVATE src/modern.cpp)
    target_compile_definitions(hand_bridge PRIVATE HAND_TOOLS_MODERN=1)
    target_link_libraries(hand_bridge PRIVATE linkerbot::cpp-sdk)
endif()
if(HAND_TOOLS_LEGACY)
    if(HAND_TOOLS_LEGACY_ROOT)
        find_package(linkerhand-legacy-cpp-sdk 2.2.1 CONFIG REQUIRED PATHS "${HAND_TOOLS_LEGACY_ROOT}" NO_DEFAULT_PATH)
    else()
        find_package(linkerhand-legacy-cpp-sdk 2.2.1 CONFIG REQUIRED)
    endif()
    target_sources(hand_bridge PRIVATE src/legacy.cpp)
    target_compile_definitions(hand_bridge PRIVATE HAND_TOOLS_LEGACY=1)
    target_link_libraries(hand_bridge PRIVATE LinkerHand::linkerhand_legacy_cpp_sdk)
    include(cmake/LegacyRuntime.cmake)
endif()
if(HAND_TOOLS_HISTORICAL_EXAMPLES)
    if(NOT HAND_TOOLS_LEGACY_ROOT)
        message(FATAL_ERROR "Historical examples require HAND_TOOLS_LEGACY_ROOT")
    endif()
    add_subdirectory(examples)
endif()
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME tools_offline COMMAND ${Python3_EXECUTABLE} -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tools.py" --bridge $<TARGET_FILE:hand_bridge>)
endif()
set(_modern_json false)
set(_legacy_json false)
if(HAND_TOOLS_MODERN)
    set(_modern_json true)
endif()
if(HAND_TOOLS_LEGACY)
    set(_legacy_json true)
endif()
file(WRITE "${CMAKE_BINARY_DIR}/bin/hand-tools-build.json" "{\"modern\":${_modern_json},\"legacy\":${_legacy_json}}\n")
install(FILES "${CMAKE_BINARY_DIR}/bin/hand-tools-build.json" DESTINATION bin)
install(TARGETS hand_bridge hand_simulator RUNTIME DESTINATION bin)
install(DIRECTORY python/hand_tools DESTINATION share/linkerbot-hand-tools/python PATTERN __pycache__ EXCLUDE)
install(FILES README.md NOTICE LICENSE DESTINATION share/linkerbot-hand-tools)
install(PROGRAMS hand-tools.py DESTINATION bin RENAME hand-tools.py)
