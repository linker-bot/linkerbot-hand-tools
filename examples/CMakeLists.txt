# Historical hardware examples are opt-in and linked against the external SDK.
if(NOT HAND_TOOLS_LEGACY)
    message(FATAL_ERROR "Historical examples require HAND_TOOLS_LEGACY=ON")
endif()
set(_history "${CMAKE_CURRENT_SOURCE_DIR}/../historical/v2.2.1/examples")
include("${_history}/sources.cmake")
set(_stems ${LINKERHAND_EXAMPLES})
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$" AND LINKERHAND_USE_CANFD)
    list(APPEND _stems ${LINKERHAND_EXAMPLES_CANFD})
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    list(APPEND _stems ${LINKERHAND_EXAMPLES_LINUX})
endif()
foreach(_stem IN LISTS _stems)
    if(EXISTS "${_history}/${_stem}.cpp")
        get_filename_component(_name "${_stem}" NAME)
        set(_target "v221_${_name}")
        add_executable(${_target} "${_history}/${_stem}.cpp")
        target_include_directories(${_target} PRIVATE "${_history}")
        target_link_libraries(${_target} PRIVATE LinkerHand::linkerhand_cpp_sdk)
        set_target_properties(${_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
        foreach(_config IN LISTS CMAKE_CONFIGURATION_TYPES)
            string(TOUPPER "${_config}" _upper)
            set_target_properties(${_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${_upper} "${CMAKE_BINARY_DIR}/bin")
        endforeach()
        if(TARGET legacy_runtime)
            add_dependencies(${_target} legacy_runtime)
        elseif(EXISTS "${_vendor_dir}")
            target_link_options(${_target} PRIVATE "-Wl,-rpath-link,${_vendor_dir}")
            set_property(TARGET ${_target} APPEND PROPERTY BUILD_RPATH "${_vendor_dir}")
        endif()
    endif()
endforeach()
